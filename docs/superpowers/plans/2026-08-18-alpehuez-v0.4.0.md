> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 AlpeHuez 从 v0.3.0 升级到 v0.4.0：Mint Light / Linear Dark 双主题、命令面板、Bento 网格、能力环形图、Telemetry Pod 抽屉、幽灵终端修复、无边框窗口等全部落地，一次发布。

#[cfg(windows)]
use std::os::windows::process::CommandExt;

/// Windows 下禁止子进程创建控制台窗口（消灭幽灵终端）。
fn silent(cmd: &mut Command) {
    #[cfg(windows)]

}

### Task 2: git_push 异步 + deploy-log 事件推送

**Files:**

- [ ] **Step 1: 改写 `git_push` 为分阶段推送事件**

```rust
#[tauri::command]
pub async fn git_push(app: tauri::AppHandle, message: String) -> Result<PushResult, String> {
    use tauri::Emitter;
    let root = repo_root();
    let msg = if message.trim().is_empty() {
        "Update site content".to_string()
    } else {
        message.trim().to_string()
    };
    let emit = |phase: &str, text: &str| {
        let _ = app.emit("deploy-log", serde_json::json!({ "phase": phase, "text": text }));
    };
    emit("add", "$ git add -A\n");
    let (_, add) = run_cmd("git", &["add", "-A"], root);
    emit("add", &add);
    emit("commit", "$ git commit -m \"...\"\n");
    let (commit_code, commit) = run_cmd("git", &["commit", "-m", &msg], root);
    emit("commit", &commit);
    emit("push", "$ git push\n");
    let (push_code, push) = run_cmd("git", &["push"], root);
    emit("push", &push);
    Ok(PushResult { ok: true, add, commit, commit_code, push, push_code })
}
```

- [ ] **Step 2: 面板块式控制台（Warp 风格）**

`panel/panel.js` 监听 `deploy-log`，每个 phase 渲染为一个独立 Block（成功泛薄荷绿边，失败泛红边）：

```js
listen('deploy-log', (e) => {
    const { phase, text } = e.payload;
    const ok = phase === 'add' || phase === 'commit' || phase === 'push';
    const block = document.createElement('div');
    block.className = 'deploy-block' + (ok ? ' ok' : ' err');
    block.innerHTML = `<div class="deploy-block-head">${escapeHtml(phase)}</div><pre>${escapeHtml(text)}</pre>`;
    logArea.appendChild(block);
    logArea.scrollTop = logArea.scrollHeight;
});

#[tauri::command]
pub async fn set_internal_page_visible(app: tauri::AppHandle, label: String, visible: bool) -> Result<(), String> {
    if let Some(win) = app.get_webview_window(&label) {
        if visible {
            win.show().map_err(|e| e.to_string())?;
            win.set_focus().map_err(|e| e.to_string())?;
        } else {
            win.hide().map_err(|e| e.to_string())?;
        }
    }
    Ok(())
}

const commandPalette = $('commandPalette');
const cpInput = $('cpInput');
const cpResults = $('cpResults');

function openCommandPalette() {
    commandPalette.hidden = false;
    cpInput.value = '';
    renderCommandPalette();
    setTimeout(() => cpInput.focus(), 30);
}
function closeCommandPalette() {
    commandPalette.hidden = true;
}
function renderCommandPalette() {
    const q = cpInput.value.trim();
    const isCmd = q.startsWith('/');
    const query = isCmd ? q.slice(1) : q;
    const items = [];
    if (!isCmd) {
        items.push({ kind: 'search', title: `Search ${ENGINES[activeEngine].label} for "${q || '...'}"`, run: () => { if (q) { trackAction(`Searched ${ENGINES[activeEngine].label} for "${q.slice(0,40)}"`); openUrl(ENGINES[activeEngine].url + encodeURIComponent(q), ENGINES[activeEngine].label); } closeCommandPalette(); } });
        allItems.filter((it) => !query || (it.title + ' ' + it.url).toLowerCase().includes(query.toLowerCase())).slice(0, 8).forEach((it) => {
            items.push({ kind: 'link', title: it.title, sub: it.url, run: () => { openLink(it, {}); closeCommandPalette(); } });
        });
    }
    const cmds = filteredCommands.filter((c) => !query || c.title.toLowerCase().includes(query.toLowerCase()));
    cmds.forEach((c) => items.push({ kind: 'command', title: c.title, run: () => { runCommand(c); closeCommandPalette(); } }));
    cpResults.innerHTML = items.length ? items.map((it, i) => `
        <div class="cp-item ${i === 0 ? 'active' : ''}" data-i="${i}">
            <span>${escapeHtml(it.title)}</span>
            ${it.sub ? `<span class="cp-kind">${escapeHtml(it.sub)}</span>` : `<span class="cp-kind">${it.kind}</span>`}
        </div>`).join('') : '<div class="cp-item" style="color:var(--text-muted)">No results</div>';
    cpResults._items = items;
}
cpInput.addEventListener('input', renderCommandPalette);
cpInput.addEventListener('keydown', (e) => {
    const items = cpResults._items || [];
    const actives = cpResults.querySelectorAll('.cp-item');
    let idx = [...actives].findIndex((el) => el.classList.contains('active'));
    if (e.key === 'ArrowDown') { e.preventDefault(); idx = (idx + 1) % actives.length; }
    else if (e.key === 'ArrowUp') { e.preventDefault(); idx = (idx - 1 + actives.length) % actives.length; }
    else if (e.key === 'Enter') { e.preventDefault(); if (items[idx]) items[idx].run(); return; }
    else if (e.key === 'Escape') { closeCommandPalette(); return; }
    actives.forEach((el, i) => el.classList.toggle('active', i === idx));
});
cpResults.addEventListener('click', (e) => {
    const item = e.target.closest('.cp-item');
    if (!item) return;
    const items = cpResults._items || [];
    const it = items[Number(item.dataset.i)];
    if (it) it.run();
});
commandPalette.addEventListener('click', (e) => { if (e.target.classList.contains('cp-backdrop')) closeCommandPalette(); });
```

async function sleepTab(label) {
    const tab = browserTabs.get(label);
    if (!tab || tab.sleeping) return;
    tab.sleeping = true;
    if (isDesktop) {
        try { await invoke('set_internal_page_visible', { label, visible: false }); } catch (e) { console.error(e); }
    }
    renderBrowserTabs();
}
async function wakeTab(label) {
    const tab = browserTabs.get(label);
    if (!tab || !tab.sleeping) return;
    tab.sleeping = false;
    if (isDesktop) {
        try { await invoke('set_internal_page_visible', { label, visible: true }); } catch (e) { console.error(e); }
    }
    renderBrowserTabs();
}

pub fn mark_first_run(app: tauri::AppHandle) -> bool {
    let marker = app

        .unwrap_or_default()
        .join("first_run.marker");
